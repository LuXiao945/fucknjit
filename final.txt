(function() {
    console.log('🕵️‍♂️ 启动元素级反向探测...');
    let targetVm = null;

    const video = document.querySelector('video');
    if (video) {
        let currentVm = video.__vue__;
        while (currentVm) {
            if (currentVm.timeUpdate && currentVm.model && currentVm.submit) {
                targetVm = currentVm;
                break;
            }
            currentVm = currentVm.$parent;
        }
    }

    if (!targetVm) {
        const modal = document.querySelector('.ant-modal-body');
        if (modal) {
            let currentVm = modal.__vue__;
            while (currentVm) {
                if (currentVm.submit) {
                    targetVm = currentVm;
                    break;
                }
                currentVm = currentVm.$parent;
            }
        }
    }

    if (!targetVm) {
        console.log('⏳ 前两种方式未找到，正在执行全局节点扫描（可能需要几秒钟）...');
        const allElements = document.querySelectorAll('*');
        for (let el of allElements) {
            if (el.__vue__) {
                let currentVm = el.__vue__;
                while (currentVm) {
                    if (currentVm.model && currentVm.model.questionRelaList !== undefined && typeof currentVm.submit === 'function') {
                        targetVm = currentVm;
                        break;
                    }
                    currentVm = currentVm.$parent;
                }
                if (targetVm) break;
            }
        }
    }

    if (targetVm) {
        window.vm = targetVm; 
        console.log('✅ 成功抓取到目标 Vue 组件！已挂载为 window.vm');
        console.log('📝 该视频绑定的题目数:', window.vm.model.questionRelaList ? window.vm.model.questionRelaList.length : 0);
        window.videoPlayer = document.querySelector('video');
        console.log('🎬 视频播放器已挂载为 window.videoPlayer');
    } else {
        console.error('❌ 彻底失败了。这说明页面可能不是 Vue2 写的，或者 `__vue__` 被构建工具（如 Vite/Webpack）屏蔽了。请确认你是否在正确的播放页面上。');
    }
})();
